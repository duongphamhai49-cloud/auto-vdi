(function () {
    'use strict';

    const SERVER = 'http://localhost:5050';
    const POLL_INTERVAL = 200; // 0.2 giây gửi poll một lần (tăng tốc 2.5x)
    let isProcessing = false;
    let badge = null;
    let currentSession = 0;

    // ==========================================
    // BADGE TRẠNG THÁI (vị trí gốc)
    // ==========================================
    function createBadge() {
        if (document.getElementById('aet-findopen-badge')) return;
        badge = document.createElement('div');
        badge.id = 'aet-findopen-badge';
        badge.setAttribute('data-html2canvas-ignore', 'true');
        badge.style.cssText = `
            position: fixed; top: 8px; right: 120px; z-index: 9999999;
            background: rgba(0,0,0,0.85); color: #3b82f6; padding: 6px 12px;
            border-radius: 20px; font-size: 12px; font-family: monospace;
            border: 1px solid #3b82f6; cursor: default;
            backdrop-filter: blur(8px); transition: all 0.3s ease;
        `;
        badge.textContent = '🔵 LMS FindOpen Ready';
        document.body.appendChild(badge);

        // Nút chụp ảnh test riêng biệt (nằm bên trái badge)
        if (!document.getElementById('aet-btn-manual-cap')) {
            const capBtn = document.createElement('button');
            capBtn.id = 'aet-btn-manual-cap';
            capBtn.setAttribute('data-html2canvas-ignore', 'true');
            capBtn.textContent = '📷 Test Chụp';
            capBtn.title = 'Bấm để chụp ảnh Canvas khu vực câu hỏi ngay lập tức và gửi về Server (không cần chạy lại từ đầu)';
            capBtn.style.cssText = `
                position: fixed; top: 7px; right: 330px; z-index: 9999999;
                background: #8b5cf6; color: #ffffff; border: 1px solid #a78bfa;
                padding: 5px 12px; border-radius: 16px; font-size: 12px; font-family: sans-serif; font-weight: bold;
                cursor: pointer; outline: none; white-space: nowrap;
                box-shadow: 0 2px 8px rgba(139, 92, 246, 0.4);
                backdrop-filter: blur(8px); transition: all 0.2s ease;
            `;
            capBtn.onmouseenter = () => { capBtn.style.background = '#7c3aed'; capBtn.style.transform = 'scale(1.05)'; };
            capBtn.onmouseleave = () => { capBtn.style.background = '#8b5cf6'; capBtn.style.transform = 'scale(1)'; };
            capBtn.onclick = async (e) => {
                e.stopPropagation();
                e.preventDefault();
                log('▶️ Người dùng bấm nút [📷 Test Chụp]!');
                capBtn.disabled = true;
                capBtn.style.opacity = '0.6';
                capBtn.textContent = '⏳ Đang chụp...';
                try {
                    await doCapture();
                    log('✅ Chụp ảnh bằng nút thủ công thành công!');
                    capBtn.textContent = '✅ Thành công!';
                    capBtn.style.background = '#10b981';
                    setTimeout(() => { capBtn.textContent = '📷 Test Chụp'; capBtn.style.background = '#8b5cf6'; }, 3000);
                } catch (err) {
                    console.error('❌ Lỗi nút chụp ảnh thủ công:', err);
                    capBtn.textContent = '❌ Lỗi!';
                    capBtn.style.background = '#ef4444';
                    setTimeout(() => { capBtn.textContent = '📷 Test Chụp'; capBtn.style.background = '#8b5cf6'; }, 3000);
                } finally {
                    capBtn.disabled = false;
                    capBtn.style.opacity = '1';
                }
            };
            document.body.appendChild(capBtn);
        }

        // Nút quét DOM riêng biệt (nằm bên trái nút Test Chụp)
        if (!document.getElementById('aet-btn-dom-dump')) {
            const domBtn = document.createElement('button');
            domBtn.id = 'aet-btn-dom-dump';
            domBtn.setAttribute('data-html2canvas-ignore', 'true');
            domBtn.textContent = '📋 Quét DOM';
            domBtn.title = 'Quét cấu trúc DOM khu vực câu hỏi và gửi về Server để lưu file dom_dump.txt';
            domBtn.style.cssText = `
                position: fixed; top: 7px; right: 450px; z-index: 9999999;
                background: #f59e0b; color: #ffffff; border: 1px solid #fbbf24;
                padding: 5px 12px; border-radius: 16px; font-size: 12px; font-family: sans-serif; font-weight: bold;
                cursor: pointer; outline: none; white-space: nowrap;
                box-shadow: 0 2px 8px rgba(245, 158, 11, 0.4);
                backdrop-filter: blur(8px); transition: all 0.2s ease;
            `;
            domBtn.onmouseenter = () => { domBtn.style.background = '#d97706'; domBtn.style.transform = 'scale(1.05)'; };
            domBtn.onmouseleave = () => { domBtn.style.background = '#f59e0b'; domBtn.style.transform = 'scale(1)'; };
            domBtn.onclick = async (e) => {
                e.stopPropagation();
                e.preventDefault();
                domBtn.disabled = true;
                domBtn.style.opacity = '0.6';
                domBtn.textContent = '⏳ Đang quét...';
                try {
                    // Hàm quét DOM tree với độ sâu giới hạn
                    function dumpEl(el, depth, maxDepth) {
                        if (!el || depth > maxDepth) return '';
                        const indent = '  '.repeat(depth);
                        const tag = el.tagName ? el.tagName.toLowerCase() : '#text';
                        if (tag === '#text') return '';
                        const id = el.id ? `#${el.id}` : '';
                        const cls = el.className && typeof el.className === 'string' ? '.' + el.className.trim().split(/\s+/).join('.') : '';
                        const w = el.offsetWidth || 0;
                        const h = el.offsetHeight || 0;
                        const vis = (w > 0 && h > 0) ? '' : ' [HIDDEN]';
                        const textSnippet = el.childNodes.length === 1 && el.childNodes[0].nodeType === 3 
                            ? ` text="${el.textContent.trim().substring(0, 50)}"` : '';
                        let result = `${indent}<${tag}${id}${cls}> (${w}x${h})${vis}${textSnippet}\n`;
                        for (const child of el.children) {
                            result += dumpEl(child, depth + 1, maxDepth);
                        }
                        return result;
                    }

                    let dump = '=== DOM DUMP (Full Scan) ===\n';
                    dump += `URL: ${location.href}\n`;
                    dump += `Thời gian: ${new Date().toLocaleString()}\n\n`;

                    // 1. Quét toàn bộ body tree (depth giới hạn = 6, bỏ qua script/style/svg/noscript)
                    dump += '--- BODY TREE (depth <= 6) ---\n';
                    const skipTags = new Set(['script', 'style', 'svg', 'noscript', 'link', 'meta', 'path', 'circle', 'rect', 'line', 'polyline', 'polygon', 'g', 'defs', 'clippath', 'use', 'symbol']);
                    function dumpTree(el, depth) {
                        if (!el || depth > 6) return '';
                        const tag = (el.tagName || '').toLowerCase();
                        if (skipTags.has(tag)) return '';
                        if (el.getAttribute && el.getAttribute('data-html2canvas-ignore') === 'true') return '';
                        const indent = '  '.repeat(depth);
                        const id = el.id ? `#${el.id}` : '';
                        const cls = el.className && typeof el.className === 'string' ? '.' + el.className.trim().split(/\s+/).slice(0, 5).join('.') : '';
                        const w = el.offsetWidth || 0;
                        const h = el.offsetHeight || 0;
                        const vis = (w > 0 && h > 0) ? '' : ' [HIDDEN]';
                        // Snippet of direct text (first text node only, max 60 chars)
                        let textSnippet = '';
                        for (const cn of el.childNodes) {
                            if (cn.nodeType === 3 && cn.textContent.trim()) {
                                textSnippet = ` text="${cn.textContent.trim().substring(0, 60)}"`;
                                break;
                            }
                        }
                        let result = `${indent}<${tag}${id}${cls}> (${w}x${h})${vis}${textSnippet}\n`;
                        for (const child of el.children) {
                            result += dumpTree(child, depth + 1);
                        }
                        return result;
                    }
                    dump += dumpTree(document.body, 0);

                    // 2. Tìm tất cả phần tử chứa text "Câu hỏi" hoặc "đáp án"
                    dump += '\n\n--- Phần tử chứa text "Câu hỏi" hoặc "đáp án" ---\n';
                    const allEls = document.body.querySelectorAll('*');
                    let matchCount = 0;
                    allEls.forEach(el => {
                        if (matchCount > 30) return;
                        const tag = (el.tagName || '').toLowerCase();
                        if (skipTags.has(tag)) return;
                        // Check direct text only (not nested)
                        let directText = '';
                        for (const cn of el.childNodes) {
                            if (cn.nodeType === 3) directText += cn.textContent;
                        }
                        directText = directText.trim();
                        if (directText && (directText.includes('Câu hỏi') || directText.includes('đáp án'))) {
                            const id = el.id ? `#${el.id}` : '';
                            const cls = el.className && typeof el.className === 'string' ? '.' + el.className.trim().split(/\s+/).join('.') : '';
                            dump += `[MATCH] <${tag}${id}${cls}> (${el.offsetWidth}x${el.offsetHeight}) directText="${directText.substring(0, 80)}"\n`;
                            // Parent chain (5 levels)
                            let p = el.parentElement;
                            let lvl = 1;
                            while (p && p !== document.body && lvl <= 5) {
                                const ptag = p.tagName.toLowerCase();
                                const pid = p.id ? `#${p.id}` : '';
                                const pcls = p.className && typeof p.className === 'string' ? '.' + p.className.trim().split(/\s+/).slice(0, 4).join('.') : '';
                                dump += `  [Parent ${lvl}] <${ptag}${pid}${pcls}> (${p.offsetWidth}x${p.offsetHeight})\n`;
                                p = p.parentElement;
                                lvl++;
                            }
                            matchCount++;
                        }
                    });
                    if (matchCount === 0) dump += '(Không tìm thấy)\n';

                    // 3. Tìm các table/tr trên trang
                    dump += '\n\n--- Tất cả TABLE và TR trên trang ---\n';
                    const tables = document.querySelectorAll('table');
                    dump += `Tables: ${tables.length}\n`;
                    tables.forEach((tbl, i) => {
                        const cls = tbl.className && typeof tbl.className === 'string' ? tbl.className : '';
                        const trs = tbl.querySelectorAll('tr');
                        dump += `[Table ${i}] class="${cls}" (${tbl.offsetWidth}x${tbl.offsetHeight}) rows=${trs.length}\n`;
                    });

                    // 4. Tìm phần tử có class chứa "question", "content", "detail"
                    dump += '\n\n--- Phần tử có class chứa "question/content/detail/expanded" ---\n';
                    let classMatchCount = 0;
                    allEls.forEach(el => {
                        if (classMatchCount > 40) return;
                        const cls = el.className && typeof el.className === 'string' ? el.className : '';
                        if (cls && (cls.includes('question') || cls.includes('content') || cls.includes('detail') || cls.includes('expanded'))) {
                            const tag = el.tagName.toLowerCase();
                            const id = el.id ? `#${el.id}` : '';
                            dump += `<${tag}${id}> class="${cls.substring(0, 100)}" (${el.offsetWidth}x${el.offsetHeight}) children=${el.children.length}\n`;
                            classMatchCount++;
                        }
                    });

                    // Gửi dump về server
                    await postJSON(`${SERVER}/lms/dom-dump`, { dump: dump });
                    
                    log('✅ Quét DOM thành công! Đã lưu vào dom_dump.txt');
                    domBtn.textContent = '✅ Đã quét!';
                    domBtn.style.background = '#10b981';
                    setTimeout(() => { domBtn.textContent = '📋 Quét DOM'; domBtn.style.background = '#f59e0b'; }, 3000);
                } catch (err) {
                    console.error('❌ Lỗi quét DOM:', err);
                    domBtn.textContent = '❌ Lỗi!';
                    domBtn.style.background = '#ef4444';
                    setTimeout(() => { domBtn.textContent = '📋 Quét DOM'; domBtn.style.background = '#f59e0b'; }, 3000);
                } finally {
                    domBtn.disabled = false;
                    domBtn.style.opacity = '1';
                }
            };
            document.body.appendChild(domBtn);
        }
    }

    function setBadge(text, color) {
        if (badge) {
            badge.textContent = text;
            badge.style.color = color;
            badge.style.borderColor = color;
        }
    }

    // ==========================================
    // TIỆN ÍCH HTTP & DOM
    // ==========================================
    function postJSON(url, data) {
        return new Promise((resolve, reject) => {
            GM_xmlhttpRequest({
                method: 'POST',
                url: url,
                headers: { 'Content-Type': 'application/json' },
                data: JSON.stringify(data),
                onload: (res) => {
                    try {
                        resolve(JSON.parse(res.responseText));
                    } catch (e) {
                        resolve({});
                    }
                },
                onerror: reject,
                ontimeout: reject,
                timeout: 10000,
            });
        });
    }

    function sleep(ms) {
        return new Promise(r => setTimeout(r, ms));
    }

    function log(msg) {
        console.log(`[LMS_FindOpen] ${msg}`);
    }

    function warn(msg) {
        console.warn(`[LMS_FindOpen] ⚠️ ${msg}`);
    }

    function triggerInputEvents(el) {
        if (!el) return;
        el.dispatchEvent(new Event('input', { bubbles: true }));
        el.dispatchEvent(new Event('change', { bubbles: true }));
        el.dispatchEvent(new Event('blur', { bubbles: true }));
    }

    function setReactValue(element, value) {
        if (!element) return;
        let lastValue = element.value;
        element.value = value;
        element.setAttribute('value', value);
        let event = new Event("input", { bubbles: true });
        let tracker = element._valueTracker;
        if (tracker) {
            tracker.setValue(lastValue);
        }
        element.dispatchEvent(event);
        element.dispatchEvent(new Event("change", { bubbles: true }));
    }

    function highlightElement(el, color = 'red') {
        if (!el) return;
        const oldBorder = el.style.border;
        el.style.border = `4px solid ${color}`;
        setTimeout(() => { el.style.border = oldBorder; }, 3000);
    }

    // ==========================================
    // XỬ LÝ LỆNH TỪ SERVER
    // ==========================================
    async function handleCommand(cmd) {
        const action = cmd.action;
        currentSession = cmd.session || 0;
        isProcessing = true;
        setBadge(`⚙️ LMS: ${action}...`, '#f59e0b');

        try {
            switch (action) {
                case 'search':
                    await doSearch(cmd.payload.id);
                    break;
                case 'capture':
                    await doCapture();
                    break;
                case 'edit':
                    await doEdit();
                    break;
                case 'check_form':
                    await doCheckForm();
                    break;
                default:
                    isProcessing = false;
                    setBadge('🔵 LMS FindOpen Ready', '#3b82f6');
                    return;
            }
        } catch (err) {
            console.error('[LMS_FindOpen] Error:', err);
            setBadge(`❌ Lỗi: ${err.message}`, '#ef4444');
            await postJSON(`${SERVER}/lms/done`, {
                event: `lms_${action}_done`,
                session: currentSession,
                error: err.message,
            });
        }

        isProcessing = false;
        setBadge('🔵 LMS FindOpen Ready', '#3b82f6');
    }

    // --- DO SEARCH (Tìm kiếm ID bằng DOM - Đã tối ưu tốc độ) ---
    async function doSearch(questionId) {
        setBadge(`🔍 Tìm ID: ${questionId}`, '#3b82f6');

        const exactInputSelector = 'div.ant-col.ant-form-item-control-wrapper:nth-of-type(2) > div.ant-form-item-control > span.ant-form-item-children > input.ant-input';
        let searchInput = document.querySelector(exactInputSelector);

        if (!searchInput) {
            const inputs = Array.from(document.querySelectorAll('input.ant-input, input[type="text"], input[type="search"]'));
            const visibleInputs = inputs.filter(el => el.offsetWidth > 0 && el.offsetHeight > 0);
            searchInput = visibleInputs.find(el => {
                const ph = (el.placeholder || '').toLowerCase();
                return ph.includes('mã') || ph.includes('tìm') || ph.includes('nhập') || ph.includes('search');
            });
            if (!searchInput && visibleInputs.length > 0) {
                searchInput = visibleInputs[0];
            }
        }

        if (!searchInput) {
            throw new Error('Không tìm thấy ô nhập ID trên giao diện LMS!');
        }

        highlightElement(searchInput, 'red');
        searchInput.focus();
        searchInput.click();
        await sleep(50);

        setReactValue(searchInput, questionId);
        searchInput.value = questionId;
        searchInput.setAttribute('value', questionId);
        triggerInputEvents(searchInput);
        await sleep(100);

        const exactSearchBtn = document.querySelector('div.form-button-groups:nth-of-type(2) > div.ui-form-control.m-t-15 > div > button.btn-primary.btn');
        let searchBtn = exactSearchBtn;
        if (!searchBtn) {
            const buttons = Array.from(document.querySelectorAll('button'));
            searchBtn = buttons.find(btn => {
                const txt = btn.textContent.trim().toLowerCase();
                return txt.includes('tìm kiếm') || txt.includes('tìm') || txt.includes('search');
            });
        }

        if (searchBtn) {
            highlightElement(searchBtn, 'blue');
            await sleep(50);
            searchBtn.click();
        } else {
            searchInput.dispatchEvent(new KeyboardEvent('keydown', { key: 'Enter', code: 'Enter', keyCode: 13, bubbles: true }));
            searchInput.dispatchEvent(new KeyboardEvent('keypress', { key: 'Enter', code: 'Enter', keyCode: 13, bubbles: true }));
            searchInput.dispatchEvent(new KeyboardEvent('keyup', { key: 'Enter', code: 'Enter', keyCode: 13, bubbles: true }));
        }

        // 1. Đợi nút Tìm kiếm đi vào trạng thái bận (Loading/Disabled) hoặc Table Spin xuất hiện (tối đa 1.5s)
        for (let i = 0; i < 15; i++) {
            const isBtnBusy = searchBtn && (
                searchBtn.disabled || 
                searchBtn.getAttribute('disabled') !== null || 
                searchBtn.classList.contains('ant-btn-loading') || 
                searchBtn.classList.contains('disabled') ||
                searchBtn.classList.contains('is-loading')
            );
            const isTableSpinning = !!document.querySelector('.ant-table-spin-holder, .ant-spin-spinning, .ant-spin, .ant-table-loading');
            if (isBtnBusy || isTableSpinning) {
                break;
            }
            await sleep(100);
        }

        // 2. Đợi NÚT TÌM KIẾM TRỞ LẠI TRẠNG THÁI BẤM ĐƯỢC (Ready / Enabled) và Bảng kết quả hết Loading
        let searchFinished = false;
        for (let i = 0; i < 200; i++) { // Chờ tối đa 20 giây (200 * 100ms)
            const currentSearchBtn = document.querySelector('div.form-button-groups:nth-of-type(2) > div.ui-form-control.m-t-15 > div > button.btn-primary.btn') || searchBtn;
            
            const isBtnDisabled = currentSearchBtn ? (
                currentSearchBtn.disabled || 
                currentSearchBtn.getAttribute('disabled') !== null || 
                currentSearchBtn.classList.contains('ant-btn-loading') || 
                currentSearchBtn.classList.contains('disabled') || 
                currentSearchBtn.classList.contains('is-loading')
            ) : false;

            const isTableSpinning = !!document.querySelector('.ant-table-spin-holder, .ant-spin-spinning, .ant-spin, .ant-table-loading');

            // Nút bấm đã trở lại trạng thái BẤM ĐƯỢC (Enabled) + Bảng không còn loading
            if (!isBtnDisabled && !isTableSpinning) {
                const hasRow = !!document.querySelector('tr.ant-table-row, table.tw-table tr:not(:first-child), a[href*="questionId"], svg.lucide-square-pen, i[aria-label="icon: edit"]');
                if (hasRow) {
                    searchFinished = true;
                    break;
                }
            }
            await sleep(100);
        }

        if (!searchFinished) {
            throw new Error('Tìm kiếm không hoàn tất (Nút tìm kiếm chưa về trạng thái bấm được hoặc bảng load quá chậm)');
        }

        // Nghỉ thêm 200ms cho DOM render ổn định
        await sleep(200);

        // Gửi phản hồi về server để báo hoàn tất tìm kiếm
        await postJSON(`${SERVER}/lms/done`, { event: 'lms_search_done', session: currentSession });
        setBadge('✅ Đã tìm thấy', '#10b981');
    }

    // --- DO CAPTURE (DOM html2canvas -> base64) ---
    async function doCapture() {
        setBadge('📷 Chụp ảnh (Chuẩn bị)...', '#8b5cf6');
        
        // Tìm target chụp ảnh theo thứ tự ưu tiên, chỉ chọn phần tử đang hiển thị thực sự
        let target = null;
        let targetSource = 'none';
        
        // Ưu tiên 1: Khung câu hỏi question-panel (chứa header + đề bài + đáp án)
        // Selector chính xác từ DOM dump: div.question-panel (580x383)
        const questionPanel = document.querySelector('div.question-panel');
        if (questionPanel && questionPanel.offsetWidth > 0 && questionPanel.offsetHeight > 0) {
            target = questionPanel;
            targetSource = 'div.question-panel';
        }
        
        // Ưu tiên 2: Nội dung câu hỏi (chỉ phần content, không có header)
        if (!target) {
            const qContent = document.querySelector('div.question-panel__content');
            if (qContent && qContent.offsetWidth > 0 && qContent.offsetHeight > 0) {
                target = qContent;
                targetSource = 'div.question-panel__content';
            }
        }

        // Ưu tiên 3: Khung chi tiết câu hỏi kiểu cũ (modal/panel/whitebox — backward compatibility)
        if (!target) {
            const detailAreas = Array.from(document.querySelectorAll('.ant-modal-body, .ant-modal, .modal_dialog_ant, div.question.mb-3, div.question.p-4, div.question, .whitebox'));
            const detailArea = detailAreas.find(el => {
                if (el.offsetWidth === 0 || el.offsetHeight === 0) return false;
                return el.querySelector('div.rounded.px-3.py-2, div.rounded:nth-of-type(1), .flex.flex-wrap.items-center, .ql-editor, img, .katex, .MathJax');
            });
            if (detailArea) {
                target = detailArea;
                targetSource = 'detail-area-legacy';
            }
        }
        
        // Ưu tiên 4: Dòng bảng kết quả tìm kiếm (tr.ant-table-row hoặc tw-table row)
        if (!target) {
            const trRow = document.querySelector('tr.ant-table-row') || document.querySelector('.tw-table tr:not(:first-child)');
            if (trRow && trRow.offsetWidth > 0 && trRow.offsetHeight > 0) {
                target = trRow;
                targetSource = 'table-row';
            }
        }
        
        // Ưu tiên 5: Khung soạn thảo Quill (khi đang ở chế độ edit)
        if (!target) {
            const qlEditor = document.querySelector('.ql-editor, .wrapper-quill');
            if (qlEditor && qlEditor.offsetWidth > 0 && qlEditor.offsetHeight > 0) {
                target = qlEditor;
                targetSource = '.ql-editor/.wrapper-quill';
            }
        }

        // Ưu tiên 6: Nội dung chính (main/content)
        if (!target) {
            const mainEl = document.querySelector('main, .main-content, #content, .content, [role="main"]');
            if (mainEl && mainEl.offsetWidth > 0 && mainEl.offsetHeight > 0) {
                target = mainEl;
                targetSource = 'main-fallback';
            }
        }

        // Cuối cùng: document.body
        if (!target) {
            target = document.body;
            targetSource = 'document.body';
        }

        // Log chi tiết target đã chọn để chẩn đoán
        const rect = target.getBoundingClientRect();
        log(`[CAPTURE DEBUG] Target: ${targetSource} | Tag: ${target.tagName} | Class: ${target.className} | offsetW: ${target.offsetWidth} | offsetH: ${target.offsetHeight} | rectW: ${Math.round(rect.width)} | rectH: ${Math.round(rect.height)}`);

        if (!currentSession) {
            currentSession = Date.now();
        }

        highlightElement(target, 'red');
        
        // 1. Cuộn phần tử vào giữa màn hình để hiển thị rõ ràng nhất (cả dọc và ngang)
        target.scrollIntoView({ behavior: 'instant', block: 'center', inline: 'center' });
        await sleep(300);

        // Tránh trùng lặp, đè chữ công thức toán (KaTeX/MathJax) khi html2canvas vẽ
        try {
            const allElements = target.querySelectorAll('*');
            allElements.forEach(el => {
                try {
                    const style = window.getComputedStyle(el);
                    const fontSize = parseFloat(style.fontSize);
                    const isHidden = (
                        style.display === 'none' ||
                        style.visibility === 'hidden' ||
                        style.opacity === '0' ||
                        fontSize === 0 ||
                        style.fontSize === '0px' ||
                        el.classList.contains('sr-only') ||
                        el.classList.contains('visually-hidden') ||
                        el.classList.contains('katex-mathml') ||
                        el.classList.contains('MathJax_Preview') ||
                        el.classList.contains('MathJax_MathML')
                    );
                    const isClippingHidden = (
                        style.position === 'absolute' &&
                        (style.width === '1px' || style.width === '0px') &&
                        (style.height === '1px' || style.height === '0px') &&
                        style.overflow === 'hidden'
                    );
                    const isClipRect = style.clip && (style.clip.includes('1px') || style.clip.includes('0px'));

                    if (isHidden || isClippingHidden || isClipRect) {
                        el.setAttribute('data-html2canvas-ignore', 'true');
                    }
                } catch (e) {}
            });
        } catch (e) {
            console.error('[LMS_FindOpen] Lỗi ẩn công thức trùng lặp:', e);
        }

        // Tạm thời giãn nở các khung cuộn để hiển thị trọn vẹn câu hỏi không bị khuyết
        const modifiedElements = [];
        try {
            let current = target;
            while (current && current !== document.body) {
                const style = window.getComputedStyle(current);
                if (style.overflowX === 'hidden' || style.overflowX === 'auto' || style.overflowX === 'scroll' ||
                    style.overflow === 'hidden' || style.overflow === 'auto' || style.overflow === 'scroll') {
                    modifiedElements.push({
                        element: current,
                        overflow: current.style.overflow,
                        overflowX: current.style.overflowX,
                        width: current.style.width
                    });
                    current.style.setProperty('overflow', 'visible', 'important');
                    current.style.setProperty('overflow-x', 'visible', 'important');
                    current.style.setProperty('width', 'auto', 'important');
                }
                current = current.parentElement;
            }

            const children = target.querySelectorAll('*');
            children.forEach(el => {
                const style = window.getComputedStyle(el);
                if (el.scrollWidth > el.clientWidth || style.overflowX === 'auto' || style.overflowX === 'scroll') {
                    modifiedElements.push({
                        element: el,
                        overflow: el.style.overflow,
                        overflowX: el.style.overflowX,
                        width: el.style.width
                    });
                    el.style.setProperty('overflow', 'visible', 'important');
                    el.style.setProperty('overflow-x', 'visible', 'important');
                    el.style.setProperty('width', 'auto', 'important');
                }
            });
        } catch (e) {
            console.error('[LMS_FindOpen] Lỗi giãn nở khung cuộn:', e);
        }

        try {
            setBadge('📷 Chụp ảnh (Đang vẽ)...', '#8b5cf6');
            log(`[CAPTURE DEBUG] Gọi html2canvas với target: ${target.tagName}.${target.className} (${target.offsetWidth}x${target.offsetHeight})`);
            const capturePromise = html2canvas(target, {
                useCORS: true,
                allowTaint: true,
                scale: 1.0,
                imageTimeout: 15000,
                logging: true, // BẬT logging html2canvas để debug trong F12 Console
                backgroundColor: '#ffffff',
            });

            // Giới hạn thời gian chạy html2canvas trong 18 giây, tránh treo trình duyệt
            const timeoutPromise = new Promise((_, reject) => {
                setTimeout(() => reject(new Error('HTML2Canvas Timeout')), 18000);
            });

            const canvas = await Promise.race([capturePromise, timeoutPromise]);
            
            log(`[CAPTURE DEBUG] html2canvas trả về canvas: ${canvas ? canvas.width + 'x' + canvas.height : 'null'}`);
            
            if (!canvas || canvas.width === 0 || canvas.height === 0) {
                throw new Error('Kích thước canvas bằng 0 (rỗng) - Target: ' + target.tagName + '.' + target.className + ' (' + target.offsetWidth + 'x' + target.offsetHeight + ')');
            }

            setBadge('📷 Chụp ảnh (Gửi Server)...', '#8b5cf6');

            const base64 = canvas.toDataURL('image/png');
            if (!base64 || base64 === 'data:,' || base64 === 'data:image/png;base64,' || base64.length < 100) {
                throw new Error('Dữ liệu ảnh html2canvas bị rỗng (data:,)');
            }

            await postJSON(`${SERVER}/lms/done`, {
                event: 'lms_capture_done',
                session: currentSession,
                image: base64,
            });
            setBadge('✅ Đã chụp', '#10b981');
            return base64;
        } catch (err) {
            warn("Lỗi doCapture: " + err.message);
            // Gửi báo lỗi về server để kích hoạt fallback ngay lập tức
            await postJSON(`${SERVER}/lms/done`, {
                event: 'lms_capture_done',
                session: currentSession,
                error: err.message,
            });
            setBadge(`❌ Lỗi chụp: ${err.message}`, '#ef4444');
            throw err;
        } finally {
            // Khôi phục lại overflow ban đầu
            modifiedElements.forEach(item => {
                try {
                    if (item.overflow) item.element.style.setProperty('overflow', item.overflow);
                    else item.element.style.removeProperty('overflow');

                    if (item.overflowX) item.element.style.setProperty('overflow-x', item.overflowX);
                    else item.element.style.removeProperty('overflow-x');

                    if (item.width) item.element.style.setProperty('width', item.width);
                    else item.element.style.removeProperty('width');
                } catch (e) {}
            });
        }
    }

    // --- DO EDIT (Click bút chì 1 & bút chì 2 bằng DOM) ---
    async function doEdit() {
        setBadge('✏️ Đang mở Edit...', '#f59e0b');

        // Cuộn toàn bộ về đầu trang/modal để hiển thị rõ ràng nút bấm và phục vụ tọa độ cứu hộ
        window.scrollTo(0, 0);
        const scrollable = document.querySelector('.question-panel, [id^="question-wrapper"], .ant-modal-body, .ant-modal, .modal_dialog_ant, div.question, .whitebox');
        if (scrollable) {
            try { scrollable.scrollTo(0, 0); } catch(e) {}
        }
        await sleep(200);

        function simulateFullClick(element) {
            if (!element) return;
            ['mousedown', 'pointerdown', 'mouseup', 'pointerup', 'click'].forEach(eventType => {
                try {
                    element.dispatchEvent(new MouseEvent(eventType, { bubbles: true, cancelable: true, buttons: 1 }));
                } catch(e) {}
            });
            try { element.click(); } catch(e) {}
        }

        // Hàm click toàn diện cho cả thẻ link <a>, svg và button
        function clickElementThoroughly(el) {
            if (!el) return;
            highlightElement(el, 'red');

            const linkEl = el.matches('a') ? el : (el.querySelector('a') || el.closest('a'));
            const svgEl = el.querySelector('svg') || (el.matches('svg') ? el : null);

            const mouseEvents = ['mouseenter', 'mouseover', 'mousedown', 'pointerdown', 'mouseup', 'pointerup', 'click'];
            for (const evt of mouseEvents) {
                try {
                    el.dispatchEvent(new MouseEvent(evt, { bubbles: true, cancelable: true, view: window, buttons: 1 }));
                } catch(e) {}
            }

            if (linkEl && linkEl !== el) {
                for (const evt of mouseEvents) {
                    try {
                        linkEl.dispatchEvent(new MouseEvent(evt, { bubbles: true, cancelable: true, view: window, buttons: 1 }));
                    } catch(e) {}
                }
            }

            if (svgEl && svgEl !== el) {
                for (const evt of mouseEvents) {
                    try {
                        svgEl.dispatchEvent(new MouseEvent(evt, { bubbles: true, cancelable: true, view: window, buttons: 1 }));
                    } catch(e) {}
                }
            }

            try { if (linkEl) linkEl.click(); } catch(e) {}
            try { el.click(); } catch(e) {}
            try { if (el.parentElement) el.parentElement.click(); } catch(e) {}
        }

        // BƯỚC 1: Click cây bút chì thứ nhất (Pencil 1 - trên dòng kết quả bảng)

        const selectors1 = [

            'table.tw-table tbody tr a[title*="cập nhật" i]',
            'table.tw-table tbody tr svg.lucide-square-pen',
            'table.tw-table tbody tr svg.lucide-pen-box',
            'table.tw-table tbody tr svg.lucide-edit',
            'table.tw-table tbody tr svg.lucide-pen-square',
            'table.tw-table tbody tr a',
            'table.tw-table tbody tr button',
            'table.tw-table tbody tr [role="button"]',
            '.tw-table tbody tr a[href*="questionId"]',
            '.tw-table tbody tr a',
            'table tbody tr a[href*="questionId"]',
            'table.tw-table tr:not(:first-child) a[href*="questionId"]',
            'table.tw-table tr:not(:first-child) a[title*="sửa" i]',
            'table.tw-table tr:not(:first-child) svg.lucide-square-pen',
            'table.tw-table tr:not(:first-child) a',
            'table.tw-table tr:not(:first-child) button',
            'tr.ant-table-row i[aria-label="icon: edit"]',
            'tr.ant-table-row a',
            'tr.ant-table-row button'
        ];

        for (const sel of selectors1) {
            const el = document.querySelector(sel);
            if (el && (el.offsetWidth > 0 || el.getClientRects().length > 0)) {
                editBtn1 = el.closest('a, button, [role="button"]') || el;
                log(`[Pencil 1] ✅ Tìm thấy nút Edit lần 1 theo selector: "${sel}" (Tag: <${editBtn1.tagName.toLowerCase()}>)`);
                break;
            }
        }

        // Fallback quét trong bảng nếu các selector trên chưa bắt được
        if (!editBtn1) {
            const tableRows = document.querySelectorAll('table.tw-table tbody tr, table.tw-table tr:not(:first-child), tr.ant-table-row');
            if (tableRows.length > 0) {
                const firstRow = tableRows[0];
                const clickables = firstRow.querySelectorAll('a, button, [role="button"], svg, span.cursor-pointer');
                for (const el of clickables) {
                    if (el.offsetWidth === 0 && el.offsetHeight === 0 && el.getClientRects().length === 0) continue;
                    const title = (el.title || el.getAttribute('title') || el.getAttribute('aria-label') || '').toLowerCase();
                    const href = (el.getAttribute('href') || '').toLowerCase();
                    const hasPen = !!el.querySelector('svg.lucide-square-pen, svg.lucide-edit, svg[data-icon="edit"]');
                    if (title.includes('sửa') || title.includes('cập nhật') || href.includes('questionid') || hasPen) {
                        editBtn1 = el.closest('a, button, [role="button"]') || el;
                        log(`[Pencil 1] ✅ Tìm thấy nút Edit lần 1 qua quét dòng đầu: Title="${title}", Href="${href}"`);
                        break;
                    }
                }
                if (!editBtn1 && clickables.length > 0) {
                    editBtn1 = firstRow.querySelector('a, button') || clickables[0];
                    log(`[Pencil 1] ⚠️ Chọn nút đầu tiên trên dòng bảng làm fallback`);
                }
            }
        }

        if (editBtn1) {
            highlightElement(editBtn1, 'blue');
            simulateFullClick(editBtn1);
            simulateFullClick(editBtn1.querySelector('svg'));
            if (editBtn1.parentElement) simulateFullClick(editBtn1.parentElement);
        } else {
            log('[Pencil 1] ⚠️ Không tìm thấy Pencil 1 trên bảng, có thể chi tiết câu hỏi đã mở sẵn. Tiếp tục tìm Pencil 2...');
        }

        await sleep(300);

        // Kiểm tra xem editor modal đã mở sẵn chưa (nếu Pencil 1 mở thẳng editor)
        if (document.querySelector('.wrapper-quill, .ql-editor, .ql-container, div.form-button-groups button.btn-primary')) {
            log("Modal soạn thảo đã được mở ngay sau Pencil 1! Bỏ qua Pencil 2.");
            await postJSON(`${SERVER}/lms/done`, { event: 'lms_edit_done', session: currentSession });
            setBadge('✅ Đã mở Edit', '#10b981');
            return;
        }

        // BƯỚC 2: Chờ và click cây bút chì thứ hai (Pencil 2) trên thanh công cụ chi tiết câu hỏi
        let editBtn2 = null;
        log("Đang chờ chi tiết câu hỏi hiển thị và tìm Pencil 2...");

        const selectors2 = [
            // 1. Selector cực kỳ chính xác từ thanh công cụ mới (.question-header-toolbar)
            '.question-header-toolbar a[title*="Chỉnh sửa"]',
            '.question-header-toolbar a[title*="Cập nhật"]',
            '.question-header-toolbar a[title*="sửa" i]',
            '.question-header-toolbar a[title*="cập nhật" i]',
            '.question-header-toolbar a[href*="questionId"]',
            '.question-header-toolbar a', // Duy nhất thẻ <a> trong toolbar là nút Sửa

            '.question-header-toolbar svg.lucide-pen-box',
            '.question-header-toolbar svg.lucide-edit',
            '.question-header-toolbar svg.lucide-pen-square',
            'div.rounded.px-3.py-2 a[href*="questionId"]',
            'div.rounded.px-3.py-2 a[title*="sửa" i]',

            'div.rounded.px-3.py-2 a',

            '.question-panel a[title*="sửa" i]',
            '.question-panel a[title*="cập nhật" i]',
            '.question-panel a[href*="questionId"]',
            'a[title*="Chỉnh sửa câu hỏi"]',
            'a[title*="Cập nhật câu hỏi"]',
            'a[title*="Chỉnh sửa" i]',
            'a[title*="Cập nhật" i]',
            '[title*="Chỉnh sửa câu hỏi"]',
            '[title*="Cập nhật câu hỏi"]',

            '.ant-modal-body i[aria-label="icon: edit"]',

            'i[aria-label="icon: edit"]',
            'svg[data-icon="edit"]',
            'svg[data-icon="pen"]'

        for (let i = 0; i < 35; i++) { // Chờ tối đa 7 giây (35 * 200ms)

            if (document.querySelector('.wrapper-quill, .ql-editor, .ql-container, div.form-button-groups button.btn-primary')) {
                log("Phát hiện form soạn thảo đã mở trong lúc chờ Pencil 2!");
                break;
            }

            // 1. Quét theo danh sách selector ưu tiên cho Pencil 2
            for (const sel of selectors2) {
                const matches = Array.from(document.querySelectorAll(sel));
                for (const el of matches) {
                    if (el === editBtn1 || (editBtn1 && editBtn1.contains(el))) continue;
                    if (el.offsetWidth === 0 && el.offsetHeight === 0 && el.getClientRects().length === 0) continue;

                    const target = el.matches('a, button, [role="button"]') ? el : (el.closest('a, button, [role="button"]') || el);
                    if (target !== editBtn1) {
                        editBtn2 = target;
                        log(`[Pencil 2] ✅ Tìm thấy nút Edit lần 2 theo selector: "${sel}" (Tag: <${editBtn2.tagName.toLowerCase()}>)`);
                        break;
                    }
                }
                if (editBtn2) break;
            }

            if (editBtn2) break;

            const containers = Array.from(document.querySelectorAll('.question-header-toolbar, div.rounded.px-3.py-2, .question-panel, [id^="question-wrapper"], .ant-modal-body, .ant-modal, .whitebox'));
            for (const container of containers) {
                if (container.offsetWidth === 0 && container.offsetHeight === 0) continue;
                const candidate = container.querySelector('a[href*="questionId"], a[title*="sửa" i], a[title*="cập nhật" i], a, svg.lucide-square-pen, svg.lucide-edit, i[aria-label="icon: edit"]');
                if (candidate && candidate !== editBtn1 && !editBtn1?.contains(candidate)) {
                    editBtn2 = candidate.closest('a, button, [role="button"]') || candidate;
                    log(`[Pencil 2] ✅ Tìm thấy nút Edit lần 2 trong container <${container.className}> (Tag: <${editBtn2.tagName.toLowerCase()}>)`);

                }

        }

        await postJSON(`${SERVER}/lms/done`, { event: 'lms_edit_done', session: currentSession });
        setBadge('✅ Đã mở Edit', '#10b981');
    }

    }

    function doTestScan() {
        clearDebugHighlights();
        
        // Hàm chấm điểm (copy từ doEdit)
        function scoreElement(el, idx, total) {
            let score = 50;
            const html = el.innerHTML || '';
            const allText = (el.textContent || '').toLowerCase() + ' ' + (el.title || '').toLowerCase() + ' ' + (el.getAttribute('aria-label') || '').toLowerCase();
            if (html.includes('edit') || html.includes('pencil') || html.includes('pen') || html.includes('lucide-square-pen') || allText.includes('sửa') || allText.includes('chỉnh sửa') || allText.includes('cập nhật') || allText.includes('edit')) score += 80;
            if (el.querySelector('svg')) score += 15;
            if (el.closest('div.flex.flex-wrap.items-center')) score += 10;
            if (idx === total - 2 && total >= 3) score += 500; // Nâng điểm cực mạnh để ưu tiên nút thứ 2 từ phải sang
            else if (idx === total - 1) score += 10;
            if (allText.includes('copy') || allText.includes('duplicate') || html.includes('copy') || html.includes('duplicate') || html.includes('copy-o')) score -= 60;
            if (allText.includes('delete') || allText.includes('xóa') || html.includes('delete') || html.includes('xóa') || html.includes('trash')) score -= 100;
            if (allText.includes('attach') || allText.includes('file') || html.includes('attach') || html.includes('dropzone')) score -= 80;
            return score;
        }

        const areas = Array.from(document.querySelectorAll('.question-panel, [id^="question-wrapper"], div.rounded.px-3.py-2, .ant-modal-body, .ant-modal, .modal_dialog_ant, div.question.mb-3, div.question.p-4, .whitebox, body'));
        const detailArea = areas.find(el => {
            if (el.offsetWidth === 0 && el.offsetHeight === 0) return false;
            return el.querySelector('div.rounded.px-3.py-2, div.rounded:nth-of-type(1), .flex.flex-wrap.items-center, .question-header-toolbar');
        });
        if (!detailArea) {
            alert('[Test Scan] Không tìm thấy vùng chi tiết câu hỏi đang hiển thị trên trang!');
            return;
        }

        const header = detailArea.querySelector('div.rounded.px-3.py-2, div.rounded:nth-of-type(1)');
        const searchArea = header || detailArea;
        const label = header ? 'H' : 'F';

        const rawCandidates = Array.from(searchArea.querySelectorAll('span.text-muted-700, span.text-primary, span.cursor-pointer, .cursor-pointer, i[aria-label="icon: edit"], div.attachment-container, div.attachment-container-dropzone, div.flex.flex-wrap.items-center > span'));
        const filteredCandidates = rawCandidates.filter(el => {
            if (el.offsetWidth === 0) return false;
            const tag = el.tagName.toLowerCase();
            return tag !== 'svg' && tag !== 'path' && tag !== 'i';
        });

        // Chỉ lấy phần tử ngoài cùng đại diện cho mỗi nút
        const finalCandidates = filteredCandidates.filter(el => {
            return !filteredCandidates.some(other => other !== el && other.contains(el));
        });

        if (finalCandidates.length === 0) {
            alert('[Test Scan] Không tìm thấy ứng viên nào sau khi lọc!');
            return;
        }

        let bestEl = null;
        let bestScore = -999;
        finalCandidates.forEach((el, idx) => {
            const score = scoreElement(el, idx, finalCandidates.length);
            console.log(`[Test Scan] ${label}#${idx}: Tag=${el.tagName}, Class=${el.className}, Score=${score}, HTML=${el.innerHTML.substring(0, 80)}`);
            highlightCandidateDebug(el, idx, label, score);
            if (score > bestScore) {
                bestScore = score;
                bestEl = el;
            }
        });

        if (bestEl) {
            highlightChosenDebug(bestEl);
            console.log(`[Test Scan] ✅ NÚT ĐƯỢC CHỌN: Score=${bestScore}, Tag=${bestEl.tagName}, Class=${bestEl.className}`);
        }

        setBadge(`🔍 Scan: ${finalCandidates.length} nút, chọn ${bestScore}pt`, '#FFAA00');
    }

    async function doCopyDOM() {
        const areas = Array.from(document.querySelectorAll('.question-panel, [id^="question-wrapper"], div.rounded.px-3.py-2, .ant-modal-body, .ant-modal, .modal_dialog_ant, div.question, .whitebox, body'));
        const detailArea = areas.find(el => {
            if (el.offsetWidth === 0 && el.offsetHeight === 0) return false;
            return el.querySelector('div.question, div.rounded.px-3.py-2, .flex.flex-wrap.items-center, .question-header-toolbar') || el.classList.contains('question');
        }) || document.querySelector('.question-header-toolbar, div.rounded.px-3.py-2, .question-panel, [id^="question-wrapper"], .ant-modal-body, .ant-modal, .modal_dialog_ant, div.question, .whitebox');

        if (!detailArea) {
            alert('Không tìm thấy vùng chi tiết câu hỏi đang hiển thị!');
            return;
        }

        const header = detailArea.querySelector('div.rounded.px-3.py-2, div.rounded:nth-of-type(1)') || detailArea;
        const domString = header.outerHTML;

        try {
            await navigator.clipboard.writeText(domString);
            alert('Đã copy DOM của vùng chức năng vào Clipboard thành công! Bạn hãy dán cho AI phân tích.');
        } catch (err) {
            // Hiển thị khung văn bản cho phép copy thủ công
            const container = document.createElement('div');
            container.id = 'aet-dom-copy-overlay';
            container.style.cssText = 'position:fixed;top:0;left:0;width:100%;height:100%;z-index:999999999;background:rgba(0,0,0,0.85);display:flex;flex-direction:column;align-items:center;justify-content:center;padding:20px;';
            
            const title = document.createElement('h3');
            title.textContent = 'Trình duyệt chặn ghi Clipboard. Hãy bấm Ctrl + C để copy đoạn mã dưới đây:';
            title.style.cssText = 'color:#fff;margin-bottom:15px;font-family:sans-serif;';
            
            const txt = document.createElement('textarea');
            txt.value = domString;
            txt.style.cssText = 'width:80%;height:60%;background:#1e1e1e;color:#00ff00;padding:15px;font-family:monospace;border:2px solid #00ff00;border-radius:6px;font-size:12px;resize:none;';
            
            const closeBtn = document.createElement('button');
            closeBtn.textContent = 'Đóng Lại';
            closeBtn.style.cssText = 'margin-top:20px;padding:10px 30px;background:#ef4444;color:#fff;border:none;border-radius:4px;font-weight:bold;cursor:pointer;';
            closeBtn.onclick = () => { container.remove(); };
            
            container.appendChild(title);
            container.appendChild(txt);
            container.appendChild(closeBtn);
            document.body.appendChild(container);
            
            txt.select();
            txt.focus();
        }
    }
