// ==UserScript==
// @name         LMS_FindOpen — Auto Search & Open Edit (DOM)
// @namespace    http://tampermonkey.net/
// @version      1.0
// @description  Tự động tìm ID câu hỏi và click mở 2 lớp bút chì Edit trên LMS bằng DOM manipulation.
// @author       Bạn
// @match        https://*.lotuslms.com/*
// @match        https://daymai.vn/*
// @grant        GM_xmlhttpRequest
// @connect      localhost
// @require      https://cdnjs.cloudflare.com/ajax/libs/html2canvas/1.4.1/html2canvas.min.js
// @run-at       document-idle
// ==/UserScript==

(function () {
    'use strict';

    const SERVER = 'http://localhost:5050';
    const POLL_INTERVAL = 500; // 0.5 giây gửi poll một lần
    let isProcessing = false;
    let badge = null;
    let currentSession = 0;

    // ==========================================
    // BADGE TRẠNG THÁI
    // ==========================================
    function createBadge() {
        if (document.getElementById('aet-findopen-badge')) return;
        badge = document.createElement('div');
        badge.id = 'aet-findopen-badge';
        badge.style.cssText = `
            position: fixed; top: 8px; right: 120px; z-index: 9999999;
            background: rgba(0,0,0,0.85); color: #3b82f6; padding: 6px 12px;
            border-radius: 20px; font-size: 12px; font-family: monospace;
            border: 1px solid #3b82f6; cursor: default;
            backdrop-filter: blur(8px); transition: all 0.3s ease;
        `;
        badge.textContent = '🔵 LMS FindOpen Ready';
        document.body.appendChild(badge);
    }

    function setBadge(text, color) {
        if (badge) {
            badge.textContent = text;
            badge.style.color = color;
            badge.style.borderColor = color;
        }
    }

    // ==========================================
    // TIỆN ÍCH HTTP & DOM
    // ==========================================
    function postJSON(url, data) {
        return new Promise((resolve, reject) => {
            GM_xmlhttpRequest({
                method: 'POST',
                url: url,
                headers: { 'Content-Type': 'application/json' },
                data: JSON.stringify(data),
                onload: (res) => {
                    try {
                        resolve(JSON.parse(res.responseText));
                    } catch (e) {
                        resolve({});
                    }
                },
                onerror: reject,
                ontimeout: reject,
                timeout: 10000,
            });
        });
    }

    function sleep(ms) {
        return new Promise(r => setTimeout(r, ms));
    }

    function log(msg) {
        console.log(`[LMS_FindOpen] ${msg}`);
    }

    function warn(msg) {
        console.warn(`[LMS_FindOpen] ⚠️ ${msg}`);
    }

    function triggerInputEvents(el) {
        if (!el) return;
        el.dispatchEvent(new Event('input', { bubbles: true }));
        el.dispatchEvent(new Event('change', { bubbles: true }));
        el.dispatchEvent(new Event('blur', { bubbles: true }));
    }

    function setReactValue(element, value) {
        if (!element) return;
        let lastValue = element.value;
        element.value = value;
        element.setAttribute('value', value);
        let event = new Event("input", { bubbles: true });
        let tracker = element._valueTracker;
        if (tracker) {
            tracker.setValue(lastValue);
        }
        element.dispatchEvent(event);
        element.dispatchEvent(new Event("change", { bubbles: true }));
    }

    function highlightElement(el, color = 'red') {
        if (!el) return;
        const oldBorder = el.style.border;
        el.style.border = `4px solid ${color}`;
        setTimeout(() => { el.style.border = oldBorder; }, 3000);
    }

    // ==========================================
    // XỬ LÝ LỆNH TỪ SERVER
    // ==========================================
    async function handleCommand(cmd) {
        const action = cmd.action;
        currentSession = cmd.session || 0;
        isProcessing = true;
        setBadge(`⚙️ LMS: ${action}...`, '#f59e0b');

        try {
            switch (action) {
                case 'search':
                    await doSearch(cmd.payload.id);
                    break;
                case 'capture':
                    await doCapture();
                    break;
                case 'edit':
                    await doEdit();
                    break;
                case 'check_form':
                    await doCheckForm();
                    break;
                default:
                    isProcessing = false;
                    setBadge('🔵 LMS FindOpen Ready', '#3b82f6');
                    return;
            }
        } catch (err) {
            console.error('[LMS_FindOpen] Error:', err);
            setBadge(`❌ Lỗi: ${err.message}`, '#ef4444');
            await postJSON(`${SERVER}/lms/done`, {
                event: `lms_${action}_done`,
                session: currentSession,
                error: err.message,
            });
        }

        isProcessing = false;
        setBadge('🔵 LMS FindOpen Ready', '#3b82f6');
    }

    // --- DO SEARCH (Tìm kiếm ID bằng DOM) ---
    async function doSearch(questionId) {
        setBadge(`🔍 Tìm ID: ${questionId}`, '#3b82f6');
        window.scrollTo(0, 0);
        await sleep(400);

        const exactInputSelector = 'div.ant-col.ant-form-item-control-wrapper:nth-of-type(2) > div.ant-form-item-control > span.ant-form-item-children > input.ant-input';
        let searchInput = document.querySelector(exactInputSelector);

        if (!searchInput) {
            const inputs = Array.from(document.querySelectorAll('input.ant-input, input[type="text"], input[type="search"]'));
            const visibleInputs = inputs.filter(el => el.offsetWidth > 0 && el.offsetHeight > 0);
            searchInput = visibleInputs.find(el => {
                const ph = (el.placeholder || '').toLowerCase();
                return ph.includes('mã') || ph.includes('tìm') || ph.includes('nhập') || ph.includes('search');
            });
            if (!searchInput && visibleInputs.length > 0) {
                searchInput = visibleInputs[0];
            }
        }

        if (!searchInput) {
            throw new Error('Không tìm thấy ô nhập ID trên giao diện LMS!');
        }

        highlightElement(searchInput, 'red');
        searchInput.focus();
        searchInput.click();
        await sleep(150);

        setReactValue(searchInput, questionId);
        searchInput.value = questionId;
        searchInput.setAttribute('value', questionId);
        triggerInputEvents(searchInput);
        await sleep(300);

        const exactSearchBtn = document.querySelector('div.form-button-groups:nth-of-type(2) > div.ui-form-control.m-t-15 > div > button.btn-primary.btn');
        let searchBtn = exactSearchBtn;
        if (!searchBtn) {
            const buttons = Array.from(document.querySelectorAll('button'));
            searchBtn = buttons.find(btn => {
                const txt = btn.textContent.trim().toLowerCase();
                return txt.includes('tìm kiếm') || txt.includes('tìm') || txt.includes('search');
            });
        }

        if (searchBtn) {
            highlightElement(searchBtn, 'blue');
            await sleep(150);
            searchBtn.click();
        } else {
            searchInput.dispatchEvent(new KeyboardEvent('keydown', { key: 'Enter', code: 'Enter', keyCode: 13, bubbles: true }));
            searchInput.dispatchEvent(new KeyboardEvent('keypress', { key: 'Enter', code: 'Enter', keyCode: 13, bubbles: true }));
            searchInput.dispatchEvent(new KeyboardEvent('keyup', { key: 'Enter', code: 'Enter', keyCode: 13, bubbles: true }));
        }

        // Đợi kết quả hiển thị trên bảng
        await sleep(800);
        let foundRow = false;
        for (let i = 0; i < 30; i++) { // Chờ tối đa 15 giây
            if (document.querySelector('tr.ant-table-row, i[aria-label="icon: edit"]')) {
                foundRow = true;
                break;
            }
            await sleep(500);
        }

        if (!foundRow) {
            throw new Error('Tìm kiếm không trả về kết quả (hoặc bảng load quá chậm)');
        }

        window.scrollTo(0, document.body.scrollHeight);
        await sleep(600);

        // Gửi phản hồi về server để báo hoàn tất tìm kiếm
        await postJSON(`${SERVER}/lms/done`, { event: 'lms_search_done', session: currentSession });
        setBadge('✅ Đã tìm thấy', '#10b981');
    }

    // --- DO CAPTURE (DOM html2canvas -> base64) ---
    async function doCapture() {
        setBadge('📷 Chụp ảnh (Chuẩn bị)...', '#8b5cf6');
        let target = document.querySelector('tr.ant-table-row');
        if (!target) {
            target = document.querySelector('.question-content, .question-detail, .content-detail, .ql-editor, [class*="question"]');
        }
        if (!target) {
            target = document.querySelector('main, .main-content, #content, .content, [role="main"]');
        }
        if (!target) {
            target = document.body;
        }

        if (!currentSession) {
            currentSession = Date.now();
        }

        highlightElement(target, 'red');
        
        // 1. Cuộn phần tử vào giữa màn hình để hiển thị rõ ràng nhất (cả dọc và ngang)
        target.scrollIntoView({ behavior: 'instant', block: 'center', inline: 'center' });
        await sleep(300);

        // Tránh trùng lặp, đè chữ công thức toán (KaTeX/MathJax) khi html2canvas vẽ
        try {
            const allElements = target.querySelectorAll('*');
            allElements.forEach(el => {
                try {
                    const style = window.getComputedStyle(el);
                    const fontSize = parseFloat(style.fontSize);
                    const isHidden = (
                        style.display === 'none' ||
                        style.visibility === 'hidden' ||
                        style.opacity === '0' ||
                        fontSize === 0 ||
                        style.fontSize === '0px' ||
                        el.classList.contains('sr-only') ||
                        el.classList.contains('visually-hidden') ||
                        el.classList.contains('katex-mathml') ||
                        el.classList.contains('MathJax_Preview') ||
                        el.classList.contains('MathJax_MathML')
                    );
                    const isClippingHidden = (
                        style.position === 'absolute' &&
                        (style.width === '1px' || style.width === '0px') &&
                        (style.height === '1px' || style.height === '0px') &&
                        style.overflow === 'hidden'
                    );
                    const isClipRect = style.clip && (style.clip.includes('1px') || style.clip.includes('0px'));

                    if (isHidden || isClippingHidden || isClipRect) {
                        el.setAttribute('data-html2canvas-ignore', 'true');
                    }
                } catch (e) {}
            });
        } catch (e) {
            console.error('[LMS_FindOpen] Lỗi ẩn công thức trùng lặp:', e);
        }

        // Tạm thời giãn nở các khung cuộn để hiển thị trọn vẹn câu hỏi không bị khuyết
        const modifiedElements = [];
        try {
            let current = target;
            while (current && current !== document.body) {
                const style = window.getComputedStyle(current);
                if (style.overflowX === 'hidden' || style.overflowX === 'auto' || style.overflowX === 'scroll' ||
                    style.overflow === 'hidden' || style.overflow === 'auto' || style.overflow === 'scroll') {
                    modifiedElements.push({
                        element: current,
                        overflow: current.style.overflow,
                        overflowX: current.style.overflowX,
                        width: current.style.width
                    });
                    current.style.setProperty('overflow', 'visible', 'important');
                    current.style.setProperty('overflow-x', 'visible', 'important');
                    current.style.setProperty('width', 'auto', 'important');
                }
                current = current.parentElement;
            }

            const children = target.querySelectorAll('*');
            children.forEach(el => {
                const style = window.getComputedStyle(el);
                if (el.scrollWidth > el.clientWidth || style.overflowX === 'auto' || style.overflowX === 'scroll') {
                    modifiedElements.push({
                        element: el,
                        overflow: el.style.overflow,
                        overflowX: el.style.overflowX,
                        width: el.style.width
                    });
                    el.style.setProperty('overflow', 'visible', 'important');
                    el.style.setProperty('overflow-x', 'visible', 'important');
                    el.style.setProperty('width', 'auto', 'important');
                }
            });
        } catch (e) {
            console.error('[LMS_FindOpen] Lỗi giãn nở khung cuộn:', e);
        }

        try {
            setBadge('📷 Chụp ảnh (Đang vẽ)...', '#8b5cf6');
            const capturePromise = html2canvas(target, {
                useCORS: true,
                allowTaint: true,
                scale: 1.0, // Chụp tỉ lệ 1-1 tăng tốc độ vẽ
                imageTimeout: 15000, // Chờ tải ảnh ngoài tối đa 15s
                logging: false,
                backgroundColor: '#ffffff',
            });

            // Giới hạn thời gian chạy html2canvas trong 18 giây, tránh treo trình duyệt
            const timeoutPromise = new Promise((_, reject) => {
                setTimeout(() => reject(new Error('HTML2Canvas Timeout')), 18000);
            });

            const canvas = await Promise.race([capturePromise, timeoutPromise]);
            
            setBadge('📷 Chụp ảnh (Gửi Server)...', '#8b5cf6');

            const base64 = canvas.toDataURL('image/png');
            await postJSON(`${SERVER}/lms/done`, {
                event: 'lms_capture_done',
                session: currentSession,
                image: base64,
            });
            setBadge('✅ Đã chụp', '#10b981');
        } catch (err) {
            warn("Lỗi doCapture: " + err.message);
            // Gửi báo lỗi về server để kích hoạt fallback ngay lập tức
            await postJSON(`${SERVER}/lms/done`, {
                event: 'lms_capture_done',
                session: currentSession,
                error: err.message,
            });
            setBadge(`❌ Lỗi chụp: ${err.message}`, '#ef4444');
        } finally {
            // Khôi phục lại overflow ban đầu
            modifiedElements.forEach(item => {
                try {
                    if (item.overflow) item.element.style.setProperty('overflow', item.overflow);
                    else item.element.style.removeProperty('overflow');

                    if (item.overflowX) item.element.style.setProperty('overflow-x', item.overflowX);
                    else item.element.style.removeProperty('overflow-x');

                    if (item.width) item.element.style.setProperty('width', item.width);
                    else item.element.style.removeProperty('width');
                } catch (e) {}
            });
        }
    }

    // --- DO EDIT (Click bút chì 1 & bút chì 2 bằng DOM) ---
    async function doEdit() {
        setBadge('✏️ Đang mở Edit...', '#f59e0b');

        // Cuộn toàn bộ về đầu trang/modal để hiển thị rõ ràng nút bấm và phục vụ tọa độ cứu hộ
        window.scrollTo(0, 0);
        const scrollable = document.querySelector('.ant-modal-body, .ant-modal, .modal_dialog_ant, div.question, .whitebox');
        if (scrollable) {
            try { scrollable.scrollTo(0, 0); } catch(e) {}
        }
        await sleep(200);

        function simulateFullClick(element) {
            if (!element) return;
            ['mousedown', 'pointerdown', 'mouseup', 'pointerup', 'click'].forEach(eventType => {
                try {
                    element.dispatchEvent(new MouseEvent(eventType, { bubbles: true, cancelable: true, buttons: 1 }));
                } catch(e) {}
            });
            try { element.click(); } catch(e) {}
        }

        // BƯỚC 1: Click cây bút chì thứ nhất (trên dòng bảng)
        let editBtn1 = document.querySelector('tr.ant-table-row i[aria-label="icon: edit"]');
        if (!editBtn1) editBtn1 = document.querySelector('i[aria-label="icon: edit"]');

        if (!editBtn1) {
            const buttons = document.querySelectorAll('button, a[role="button"]');
            for (const btn of buttons) {
                const svg = btn.querySelector('svg[data-icon="edit"], svg[data-icon="pen"]');
                if (svg || btn.title?.toLowerCase().includes('sửa')) {
                    editBtn1 = btn;
                    break;
                }
            }
        }

        if (!editBtn1) {
            throw new Error('Không tìm thấy cây bút chì mở chi tiết câu hỏi (Pencil 1)');
        }

        highlightElement(editBtn1, 'blue');
        simulateFullClick(editBtn1);
        simulateFullClick(editBtn1.querySelector('svg'));
        if (editBtn1.parentElement) simulateFullClick(editBtn1.parentElement);

        // BƯỚC 2: Chờ và click cây bút chì thứ hai (Pencil 2) để mở modal soạn thảo (Edit)
        let editBtn2 = null;
        log("Đang chờ chi tiết câu hỏi hiển thị và tìm Pencil 2...");

        function scoreElement(el, index, totalCount) {
            if (!el) return -999;
            let score = 0;
            const html = el.innerHTML.toLowerCase();
            const className = (typeof el.className === 'string' ? el.className : (el.className?.baseVal || '')).toLowerCase();
            
            // Lấy tất cả thuộc tính có khả năng chứa text tooltip
            const titleAttr = (el.getAttribute('title') || el.title || '').toLowerCase();
            const tooltipAttr = (el.getAttribute('data-original-title') || el.getAttribute('data-tooltip') || el.getAttribute('data-tip') || '').toLowerCase();
            const ariaAttr = (el.getAttribute('aria-label') || '').toLowerCase();
            const parentText = el.parentElement ? (el.parentElement.title || el.parentElement.getAttribute('aria-label') || el.parentElement.getAttribute('data-original-title') || '').toLowerCase() : '';
            
            const allText = `${titleAttr} ${tooltipAttr} ${ariaAttr} ${className} ${parentText}`.toLowerCase();
            
            // 1. Dấu hiệu văn bản edit/sửa cụ thể
            if (allText.includes('chỉnh sửa câu hỏi') || allText.includes('sửa câu hỏi') || allText.includes('edit question')) {
                score += 1000; // Thắng tuyệt đối nếu khớp tooltip cụ thể!
            } else if (allText.includes('edit') || allText.includes('sửa')) {
                score += 100;
            }
            
            // 2. Dấu hiệu hình ảnh bút chì
            if (html.includes('data-icon="edit"') || html.includes('anticon-edit') || html.includes('pencil') || html.includes('pen-to-square') || html.includes('edit-pencil')) {
                score += 80;
            }
            
            // 3. Vị trí: thứ 2 từ phải sang (kế cuối)
            if (totalCount >= 2 && index === totalCount - 2) {
                score += 500; // Nâng điểm cực mạnh để ưu tiên nút thứ 2 từ phải sang
            }
            
            // Trừ điểm các nút có dấu hiệu Copy
            if (allText.includes('copy') || allText.includes('duplicate') || html.includes('copy') || html.includes('duplicate') || html.includes('copy-o')) {
                score -= 60;
            }
            
            // Trừ điểm các nút có dấu hiệu Xóa/Delete
            if (allText.includes('delete') || allText.includes('xóa') || html.includes('delete') || html.includes('xóa') || html.includes('trash')) {
                score -= 100;
            }
            
            // Trừ điểm các nút đính kèm/attachment (Nút 4, 5 trong danh sách của user)
            if (allText.includes('attach') || allText.includes('file') || html.includes('attach') || html.includes('dropzone')) {
                score -= 80;
            }
            
            return score;
        }

        // Bảng màu debug để khoanh viền từng ứng viên (mỗi nút 1 màu khác nhau)
        const debugColors = ['#FF0000', '#00FF00', '#0000FF', '#FF00FF', '#FFAA00', '#00FFFF', '#FF6600', '#9900FF', '#00FF99', '#FF3366'];
        
        function highlightCandidate(el, idx, label) {
            if (!el) return;
            const color = debugColors[idx % debugColors.length];
            el.style.outline = `3px solid ${color}`;
            el.style.outlineOffset = '2px';
            // Thêm nhãn số thứ tự
            const badge = document.createElement('span');
            badge.textContent = `${label}#${idx}`;
            badge.style.cssText = `position:absolute;top:-18px;left:0;background:${color};color:#fff;font-size:10px;padding:1px 4px;border-radius:3px;z-index:99999;pointer-events:none;font-weight:bold;`;
            el.style.position = el.style.position || 'relative';
            el.appendChild(badge);
            // Tự xóa sau 8 giây
            setTimeout(() => {
                el.style.outline = '';
                el.style.outlineOffset = '';
                try { badge.remove(); } catch(e) {}
            }, 8000);
        }
        
        function highlightChosen(el) {
            if (!el) return;
            el.style.outline = '4px solid #00FF00';
            el.style.outlineOffset = '3px';
            el.style.boxShadow = '0 0 12px #00FF00';
            setTimeout(() => {
                el.style.outline = '';
                el.style.outlineOffset = '';
                el.style.boxShadow = '';
            }, 8000);
        }

        for (let i = 0; i < 25; i++) { // Chờ tối đa 5 giây (25 * 200ms)
            // Lấy vùng chi tiết đang mở (chọn modal/khung chứa đề bài và các nút chức năng đang hiển thị thực tế)
            const areas = Array.from(document.querySelectorAll('.ant-modal-body, .ant-modal, .modal_dialog_ant, div.question.mb-3, div.question.p-4, .whitebox'));
            const detailArea = areas.find(el => {
                if (el.offsetWidth === 0 && el.offsetHeight === 0) return false;
                return el.querySelector('div.rounded.px-3.py-2, div.rounded:nth-of-type(1), .flex.flex-wrap.items-center');
            });
            if (detailArea) {
                // Ưu tiên tìm trong thanh header metadata chứa các tác vụ ở đầu
                const header = detailArea.querySelector('div.rounded.px-3.py-2, div.rounded:nth-of-type(1)');
                if (header) {
                    const rawCandidates = Array.from(header.querySelectorAll('span.text-muted-700, span.text-primary, span.cursor-pointer, .cursor-pointer, i[aria-label="icon: edit"], div.attachment-container, div.attachment-container-dropzone, div.flex.flex-wrap.items-center > span'));
                    
                    // Loại bỏ các thẻ đồ họa trực tiếp
                    const filteredCandidates = rawCandidates.filter(el => {
                        if (el === editBtn1 || el.offsetWidth === 0) return false;
                        const tag = el.tagName.toLowerCase();
                        return tag !== 'svg' && tag !== 'path' && tag !== 'i';
                    });
                    
                    // Loại bỏ phần tử con trùng lặp để chỉ giữ lại phần tử cha ngoài cùng đại diện cho mỗi nút
                    const finalCandidates = filteredCandidates.filter(el => {
                        return !filteredCandidates.some(other => other !== el && other.contains(el));
                    });
                    
                    let bestEl = null;
                    let bestScore = -999;
                    finalCandidates.forEach((el, idx) => {
                        const score = scoreElement(el, idx, finalCandidates.length);
                        log(`[Pencil 2 Scan] Candidate [${idx}/${finalCandidates.length}]: Tag = ${el.tagName}, Class = ${el.className}, Score = ${score}`);
                        if (score > bestScore) {
                            bestScore = score;
                            bestEl = el;
                        }
                    });
                    
                    if (bestEl && bestScore >= 0) {
                        editBtn2 = bestEl;
                        highlightChosen(editBtn2);
                        log(`[Pencil 2 Scan] Đã chọn nút có điểm cao nhất: ${bestScore} (Tag = ${editBtn2.tagName}, Class = ${editBtn2.className})`);
                        break;
                    }
                } else {
                    // Fallback nếu không tìm thấy header riêng biệt
                    const rawCandidates = Array.from(detailArea.querySelectorAll('span.text-muted-700, span.text-primary, span.cursor-pointer, .cursor-pointer, i[aria-label="icon: edit"], div.attachment-container, div.attachment-container-dropzone, div.flex.flex-wrap.items-center > span'));
                    const filteredCandidates = rawCandidates.filter(el => {
                        if (el === editBtn1 || el.offsetWidth === 0) return false;
                        const tag = el.tagName.toLowerCase();
                        return tag !== 'svg' && tag !== 'path' && tag !== 'i';
                    });
                    
                    const finalCandidates = filteredCandidates.filter(el => {
                        return !filteredCandidates.some(other => other !== el && other.contains(el));
                    });
                    
                    let bestEl = null;
                    let bestScore = -999;
                    finalCandidates.forEach((el, idx) => {
                        const score = scoreElement(el, idx, finalCandidates.length);
                        log(`[Pencil 2 Fallback Scan] Candidate [${idx}/${finalCandidates.length}]: Tag = ${el.tagName}, Class = ${el.className}, Score = ${score}`);
                        if (score > bestScore) {
                            bestScore = score;
                            bestEl = el;
                        }
                    });
                    
                    if (bestEl && bestScore >= 40) {
                        editBtn2 = bestEl;
                        highlightChosen(editBtn2);
                        log(`[Pencil 2 Fallback Scan] Đã chọn nút có điểm cao nhất: ${bestScore} (Tag = ${editBtn2.tagName}, Class = ${editBtn2.className})`);
                        break;
                    }
                }
            }
            if (editBtn2) break;
            await sleep(200); // Tăng tốc quét từ 500ms xuống 200ms
        }

        if (editBtn2) {
            log("Tìm thấy Pencil 2 (chỉnh sửa tổng), tiến hành click...");
            highlightElement(editBtn2, 'red');
            simulateFullClick(editBtn2);
            
            // Bấm thêm SVG bên trong nếu có
            const svg = editBtn2.querySelector('svg');
            if (svg) simulateFullClick(svg);
            
            if (editBtn2.parentElement) simulateFullClick(editBtn2.parentElement);
            await sleep(200); // Giảm độ trễ cứng sau click từ 1.5s xuống 200ms
        } else {
            log("Không tìm thấy Pencil 2 tiêu chuẩn, thực hiện click fallback tất cả các bút khác...");
            let pencils = document.querySelectorAll('i[aria-label="icon: edit"]');
            for (let p of pencils) {
                if (p !== editBtn1 && p.offsetWidth > 0) {
                    simulateFullClick(p);
                    if (p.parentElement) simulateFullClick(p.parentElement);
                }
            }
            await sleep(200); // Giảm độ trễ cứng xuống 200ms
        }

        // Đợi modal soạn thảo hiển thị hoàn tất (hỗ trợ .wrapper-quill, .ql-editor, .ql-container)
        let formFound = false;
        log("Đang chờ modal soạn thảo load xong...");
        for (let i = 0; i < 50; i++) { // Chờ tối đa 5s (50 * 100ms)
            if (document.querySelector('.wrapper-quill, .ql-editor, .ql-container, div.form-button-groups button.btn-primary')) {
                formFound = true;
                break;
            }
            await sleep(100); // Tăng tần suất kiểm tra từ 500ms xuống 100ms để phản hồi ngay khi form xuất hiện
        }

        if (!formFound) {
            throw new Error('Không mở được modal soạn thảo (Không tìm thấy .wrapper-quill, .ql-editor hoặc nút cập nhật sau 5s)');
        }

        // Báo cáo về server
        await postJSON(`${SERVER}/lms/done`, { event: 'lms_edit_done', session: currentSession });
        setBadge('✅ Đã mở Edit', '#10b981');
    }

    // --- DO CHECK FORM (Cứu hộ kiểm tra trạng thái form mở) ---
    async function doCheckForm() {
        const formFound = !!document.querySelector('.wrapper-quill, .ql-editor, .ql-container, div.form-button-groups button.btn-primary');
        if (formFound) {
            await postJSON(`${SERVER}/lms/done`, { event: 'lms_check_form_done', session: currentSession });
            setBadge('✅ Đã mở Edit (Cứu hộ)', '#10b981');
        } else {
            throw new Error('Form soạn thảo vẫn chưa được mở');
        }
    }

    // ==========================================
    // POLLING LOOP
    // ==========================================
    async function pollServer() {
        if (isProcessing) return;
        try {
            const res = await postJSON(`${SERVER}/lms/poll?tool=findopen`, {});
            if (res && res.command) {
                await handleCommand(res.command);
            }
        } catch (err) {
            if (badge) setBadge('🔴 Mất kết nối', '#ef4444');
        }
    }

    // ==========================================
    // DEBUG: QUÉT VÀ KHOANH MÀU CÁC NÚT ỨNG VIÊN
    // ==========================================
    const debugColors = ['#FF0000', '#00FF00', '#0000FF', '#FF00FF', '#FFAA00', '#00FFFF', '#FF6600', '#9900FF', '#00FF99', '#FF3366'];
    let debugBadges = []; // Lưu lại các badge để xóa khi test lại

    function clearDebugHighlights() {
        debugBadges.forEach(b => { try { b.badge.remove(); b.el.style.outline = ''; b.el.style.outlineOffset = ''; b.el.style.boxShadow = ''; } catch(e) {} });
        debugBadges = [];
    }

    function highlightCandidateDebug(el, idx, label, score) {
        if (!el) return;
        const color = debugColors[idx % debugColors.length];
        el.style.outline = `3px solid ${color}`;
        el.style.outlineOffset = '2px';
        const b = document.createElement('div');
        b.textContent = `${label}#${idx} (${score}pt)`;
        b.style.cssText = `position:fixed;z-index:99999999;background:${color};color:#fff;font-size:11px;padding:2px 6px;border-radius:4px;pointer-events:none;font-weight:bold;white-space:nowrap;`;
        const rect = el.getBoundingClientRect();
        b.style.top = (rect.top - 20) + 'px';
        b.style.left = rect.left + 'px';
        document.body.appendChild(b);
        
        const item = { badge: b, el: el };
        debugBadges.push(item);
        
        // Tự động xóa sau 8 giây
        setTimeout(() => {
            el.style.outline = '';
            el.style.outlineOffset = '';
            try { b.remove(); } catch(e) {}
            const index = debugBadges.indexOf(item);
            if (index > -1) debugBadges.splice(index, 1);
        }, 8000);
    }

    function highlightChosenDebug(el) {
        if (!el) return;
        el.style.outline = '4px solid #00FF00';
        el.style.outlineOffset = '3px';
        el.style.boxShadow = '0 0 15px #00FF00';
        
        setTimeout(() => {
            el.style.outline = '';
            el.style.outlineOffset = '';
            el.style.boxShadow = '';
        }, 8000);
    }

    function doTestScan() {
        clearDebugHighlights();
        
        // Hàm chấm điểm (copy từ doEdit)
        function scoreElement(el, idx, total) {
            let score = 50;
            const html = el.innerHTML || '';
            const allText = (el.textContent || '').toLowerCase() + ' ' + (el.title || '').toLowerCase() + ' ' + (el.getAttribute('aria-label') || '').toLowerCase();
            if (html.includes('edit') || html.includes('pencil') || html.includes('pen') || allText.includes('sửa') || allText.includes('chỉnh sửa') || allText.includes('edit')) score += 80;
            if (el.querySelector('svg')) score += 15;
            if (el.closest('div.flex.flex-wrap.items-center')) score += 10;
            if (idx === total - 2 && total >= 3) score += 500; // Nâng điểm cực mạnh để ưu tiên nút thứ 2 từ phải sang
            else if (idx === total - 1) score += 10;
            if (allText.includes('copy') || allText.includes('duplicate') || html.includes('copy') || html.includes('duplicate') || html.includes('copy-o')) score -= 60;
            if (allText.includes('delete') || allText.includes('xóa') || html.includes('delete') || html.includes('xóa') || html.includes('trash')) score -= 100;
            if (allText.includes('attach') || allText.includes('file') || html.includes('attach') || html.includes('dropzone')) score -= 80;
            return score;
        }

        const areas = Array.from(document.querySelectorAll('.ant-modal-body, .ant-modal, .modal_dialog_ant, div.question.mb-3, div.question.p-4, .whitebox'));
        const detailArea = areas.find(el => {
            if (el.offsetWidth === 0 && el.offsetHeight === 0) return false;
            return el.querySelector('div.rounded.px-3.py-2, div.rounded:nth-of-type(1), .flex.flex-wrap.items-center');
        });
        if (!detailArea) {
            alert('[Test Scan] Không tìm thấy vùng chi tiết câu hỏi đang hiển thị trên trang!');
            return;
        }

        const header = detailArea.querySelector('div.rounded.px-3.py-2, div.rounded:nth-of-type(1)');
        const searchArea = header || detailArea;
        const label = header ? 'H' : 'F';

        const rawCandidates = Array.from(searchArea.querySelectorAll('span.text-muted-700, span.text-primary, span.cursor-pointer, .cursor-pointer, i[aria-label="icon: edit"], div.attachment-container, div.attachment-container-dropzone, div.flex.flex-wrap.items-center > span'));
        const filteredCandidates = rawCandidates.filter(el => {
            if (el.offsetWidth === 0) return false;
            const tag = el.tagName.toLowerCase();
            return tag !== 'svg' && tag !== 'path' && tag !== 'i';
        });

        // Chỉ lấy phần tử ngoài cùng đại diện cho mỗi nút
        const finalCandidates = filteredCandidates.filter(el => {
            return !filteredCandidates.some(other => other !== el && other.contains(el));
        });

        if (finalCandidates.length === 0) {
            alert('[Test Scan] Không tìm thấy ứng viên nào sau khi lọc!');
            return;
        }

        let bestEl = null;
        let bestScore = -999;
        finalCandidates.forEach((el, idx) => {
            const score = scoreElement(el, idx, finalCandidates.length);
            console.log(`[Test Scan] ${label}#${idx}: Tag=${el.tagName}, Class=${el.className}, Score=${score}, HTML=${el.innerHTML.substring(0, 80)}`);
            highlightCandidateDebug(el, idx, label, score);
            if (score > bestScore) {
                bestScore = score;
                bestEl = el;
            }
        });

        if (bestEl) {
            highlightChosenDebug(bestEl);
            console.log(`[Test Scan] ✅ NÚT ĐƯỢC CHỌN: Score=${bestScore}, Tag=${bestEl.tagName}, Class=${bestEl.className}`);
        }

        setBadge(`🔍 Scan: ${finalCandidates.length} nút, chọn ${bestScore}pt`, '#FFAA00');
    }

    async function doCopyDOM() {
        const areas = Array.from(document.querySelectorAll('.ant-modal-body, .ant-modal, .modal_dialog_ant, div.question, .whitebox'));
        const detailArea = areas.find(el => {
            if (el.offsetWidth === 0 && el.offsetHeight === 0) return false;
            return el.querySelector('div.question, div.rounded.px-3.py-2, .flex.flex-wrap.items-center') || el.classList.contains('question');
        }) || document.querySelector('.ant-modal-body, .ant-modal, .modal_dialog_ant, div.question, .whitebox');

        if (!detailArea) {
            alert('Không tìm thấy vùng chi tiết câu hỏi đang hiển thị!');
            return;
        }

        const header = detailArea.querySelector('div.rounded.px-3.py-2, div.rounded:nth-of-type(1)') || detailArea;
        const domString = header.outerHTML;

        try {
            await navigator.clipboard.writeText(domString);
            alert('Đã copy DOM của vùng chức năng vào Clipboard thành công! Bạn hãy dán cho AI phân tích.');
        } catch (err) {
            // Hiển thị khung văn bản cho phép copy thủ công
            const container = document.createElement('div');
            container.id = 'aet-dom-copy-overlay';
            container.style.cssText = 'position:fixed;top:0;left:0;width:100%;height:100%;z-index:999999999;background:rgba(0,0,0,0.85);display:flex;flex-direction:column;align-items:center;justify-content:center;padding:20px;';
            
            const title = document.createElement('h3');
            title.textContent = 'Trình duyệt chặn ghi Clipboard. Hãy bấm Ctrl + C để copy đoạn mã dưới đây:';
            title.style.cssText = 'color:#fff;margin-bottom:15px;font-family:sans-serif;';
            
            const txt = document.createElement('textarea');
            txt.value = domString;
            txt.style.cssText = 'width:80%;height:60%;background:#1e1e1e;color:#00ff00;padding:15px;font-family:monospace;border:2px solid #00ff00;border-radius:6px;font-size:12px;resize:none;';
            
            const closeBtn = document.createElement('button');
            closeBtn.textContent = 'Đóng Lại';
            closeBtn.style.cssText = 'margin-top:20px;padding:10px 30px;background:#ef4444;color:#fff;border:none;border-radius:4px;font-weight:bold;cursor:pointer;';
            closeBtn.onclick = () => { container.remove(); };
            
            container.appendChild(title);
            container.appendChild(txt);
            container.appendChild(closeBtn);
            document.body.appendChild(container);
            
            txt.select();
            txt.focus();
        }
    }
