// ==UserScript==
// @name         LMS_FindOpen — Auto Search & Open Edit (DOM)
// @namespace    http://tampermonkey.net/
// @version      1.0
// @description  Tự động tìm ID câu hỏi và click mở 2 lớp bút chì Edit trên LMS bằng DOM manipulation.
// @author       Bạn
// @match        https://*.lotuslms.com/*
// @match        https://daymai.vn/*
// @grant        GM_xmlhttpRequest
// @connect      localhost
// @require      https://cdnjs.cloudflare.com/ajax/libs/html2canvas/1.4.1/html2canvas.min.js
// @run-at       document-idle
// ==/UserScript==

(function () {
    'use strict';

    const SERVER = 'http://localhost:5050';
    const POLL_INTERVAL = 500; // 0.5 giây gửi poll một lần
    let isProcessing = false;
    let badge = null;
    let currentSession = 0;

    // ==========================================
    // BADGE TRẠNG THÁI
    // ==========================================
    function createBadge() {
        if (document.getElementById('aet-findopen-badge')) return;
        badge = document.createElement('div');
        badge.id = 'aet-findopen-badge';
        badge.style.cssText = `
            position: fixed; top: 8px; right: 120px; z-index: 9999999;
            background: rgba(0,0,0,0.85); color: #3b82f6; padding: 6px 12px;
            border-radius: 20px; font-size: 12px; font-family: monospace;
            border: 1px solid #3b82f6; cursor: default;
            backdrop-filter: blur(8px); transition: all 0.3s ease;
        `;
        badge.textContent = '🔵 LMS FindOpen Ready';
        document.body.appendChild(badge);
    }

    function setBadge(text, color) {
        if (badge) {
            badge.textContent = text;
            badge.style.color = color;
            badge.style.borderColor = color;
        }
    }

    // ==========================================
    // TIỆN ÍCH HTTP & DOM
    // ==========================================
    function postJSON(url, data) {
        return new Promise((resolve, reject) => {
            GM_xmlhttpRequest({
                method: 'POST',
                url: url,
                headers: { 'Content-Type': 'application/json' },
                data: JSON.stringify(data),
                onload: (res) => {
                    try {
                        resolve(JSON.parse(res.responseText));
                    } catch (e) {
                        resolve({});
                    }
                },
                onerror: reject,
                ontimeout: reject,
                timeout: 10000,
            });
        });
    }

    function sleep(ms) {
        return new Promise(r => setTimeout(r, ms));
    }

    function log(msg) {
        console.log(`[LMS_FindOpen] ${msg}`);
    }

    function warn(msg) {
        console.warn(`[LMS_FindOpen] ⚠️ ${msg}`);
    }

    function triggerInputEvents(el) {
        if (!el) return;
        el.dispatchEvent(new Event('input', { bubbles: true }));
        el.dispatchEvent(new Event('change', { bubbles: true }));
        el.dispatchEvent(new Event('blur', { bubbles: true }));
    }

    function setReactValue(element, value) {
        if (!element) return;
        let lastValue = element.value;
        element.value = value;
        element.setAttribute('value', value);
        let event = new Event("input", { bubbles: true });
        let tracker = element._valueTracker;
        if (tracker) {
            tracker.setValue(lastValue);
        }
        element.dispatchEvent(event);
        element.dispatchEvent(new Event("change", { bubbles: true }));
    }

    function highlightElement(el, color = 'red') {
        if (!el) return;
        const oldBorder = el.style.border;
        el.style.border = `4px solid ${color}`;
        setTimeout(() => { el.style.border = oldBorder; }, 3000);
    }

    // ==========================================
    // XỬ LÝ LỆNH TỪ SERVER
    // ==========================================
    async function handleCommand(cmd) {
        const action = cmd.action;
        currentSession = cmd.session || 0;
        isProcessing = true;
        setBadge(`⚙️ LMS: ${action}...`, '#f59e0b');

        try {
            switch (action) {
                case 'search':
                    await doSearch(cmd.payload.id);
                    break;
                case 'capture':
                    await doCapture();
                    break;
                case 'edit':
                    await doEdit();
                    break;
                default:
                    isProcessing = false;
                    setBadge('🔵 LMS FindOpen Ready', '#3b82f6');
                    return;
            }
        } catch (err) {
            console.error('[LMS_FindOpen] Error:', err);
            setBadge(`❌ Lỗi: ${err.message}`, '#ef4444');
            await postJSON(`${SERVER}/lms/done`, {
                event: `lms_${action}_done`,
                session: currentSession,
                error: err.message,
            });
        }

        isProcessing = false;
        setBadge('🔵 LMS FindOpen Ready', '#3b82f6');
    }

    // --- DO SEARCH (Tìm kiếm ID bằng DOM) ---
    async function doSearch(questionId) {
        setBadge(`🔍 Tìm ID: ${questionId}`, '#3b82f6');
        window.scrollTo(0, 0);
        await sleep(400);

        const exactInputSelector = 'div.ant-col.ant-form-item-control-wrapper:nth-of-type(2) > div.ant-form-item-control > span.ant-form-item-children > input.ant-input';
        let searchInput = document.querySelector(exactInputSelector);

        if (!searchInput) {
            const inputs = Array.from(document.querySelectorAll('input.ant-input, input[type="text"], input[type="search"]'));
            const visibleInputs = inputs.filter(el => el.offsetWidth > 0 && el.offsetHeight > 0);
            searchInput = visibleInputs.find(el => {
                const ph = (el.placeholder || '').toLowerCase();
                return ph.includes('mã') || ph.includes('tìm') || ph.includes('nhập') || ph.includes('search');
            });
            if (!searchInput && visibleInputs.length > 0) {
                searchInput = visibleInputs[0];
            }
        }

        if (!searchInput) {
            throw new Error('Không tìm thấy ô nhập ID trên giao diện LMS!');
        }

        highlightElement(searchInput, 'red');
        searchInput.focus();
        searchInput.click();
        await sleep(150);

        setReactValue(searchInput, questionId);
        searchInput.value = questionId;
        searchInput.setAttribute('value', questionId);
        triggerInputEvents(searchInput);
        await sleep(300);

        const exactSearchBtn = document.querySelector('div.form-button-groups:nth-of-type(2) > div.ui-form-control.m-t-15 > div > button.btn-primary.btn');
        let searchBtn = exactSearchBtn;
        if (!searchBtn) {
            const buttons = Array.from(document.querySelectorAll('button'));
            searchBtn = buttons.find(btn => {
                const txt = btn.textContent.trim().toLowerCase();
                return txt.includes('tìm kiếm') || txt.includes('tìm') || txt.includes('search');
            });
        }

        if (searchBtn) {
            highlightElement(searchBtn, 'blue');
            await sleep(150);
            searchBtn.click();
        } else {
            searchInput.dispatchEvent(new KeyboardEvent('keydown', { key: 'Enter', code: 'Enter', keyCode: 13, bubbles: true }));
            searchInput.dispatchEvent(new KeyboardEvent('keypress', { key: 'Enter', code: 'Enter', keyCode: 13, bubbles: true }));
            searchInput.dispatchEvent(new KeyboardEvent('keyup', { key: 'Enter', code: 'Enter', keyCode: 13, bubbles: true }));
        }

        // Đợi kết quả hiển thị trên bảng
        await sleep(800);
        let foundRow = false;
        for (let i = 0; i < 30; i++) { // Chờ tối đa 15 giây
            if (document.querySelector('tr.ant-table-row, i[aria-label="icon: edit"]')) {
                foundRow = true;
                break;
            }
            await sleep(500);
        }

        if (!foundRow) {
            throw new Error('Tìm kiếm không trả về kết quả (hoặc bảng load quá chậm)');
        }

        window.scrollTo(0, document.body.scrollHeight);
        await sleep(600);

        // Gửi phản hồi về server để báo hoàn tất tìm kiếm
        await postJSON(`${SERVER}/lms/done`, { event: 'lms_search_done', session: currentSession });
        setBadge('✅ Đã tìm thấy', '#10b981');
    }

    // --- DO CAPTURE (DOM html2canvas -> base64) ---
    async function doCapture() {
        setBadge('📷 Chụp ảnh...', '#8b5cf6');
        let target = document.querySelector('tr.ant-table-row');
        if (!target) {
            target = document.querySelector('.question-content, .question-detail, .content-detail, .ql-editor, [class*="question"]');
        }
        if (!target) {
            target = document.querySelector('main, .main-content, #content, .content, [role="main"]');
        }
        if (!target) {
            target = document.body;
        }

        highlightElement(target, 'red');
        await sleep(300);

        try {
            const canvas = await html2canvas(target, {
                useCORS: true,
                allowTaint: true,
                scale: 1.2, // Tăng nhẹ độ phân giải ảnh chụp
                logging: false,
                backgroundColor: '#ffffff',
            });

            const base64 = canvas.toDataURL('image/png');
            await postJSON(`${SERVER}/lms/done`, {
                event: 'lms_capture_done',
                session: currentSession,
                image: base64,
            });
            setBadge('✅ Đã chụp', '#10b981');
        } catch (err) {
            warn("Lỗi html2canvas: " + err.message);
            throw err;
        }
    }

    // --- DO EDIT (Click bút chì 1 & bút chì 2 bằng DOM) ---
    async function doEdit() {
        setBadge('✏️ Đang mở Edit...', '#f59e0b');

        function simulateFullClick(element) {
            if (!element) return;
            ['mousedown', 'pointerdown', 'mouseup', 'pointerup', 'click'].forEach(eventType => {
                try {
                    element.dispatchEvent(new MouseEvent(eventType, { bubbles: true, cancelable: true, buttons: 1 }));
                } catch(e) {}
            });
            try { element.click(); } catch(e) {}
        }

        // BƯỚC 1: Click cây bút chì thứ nhất (trên dòng bảng)
        let editBtn1 = document.querySelector('tr.ant-table-row i[aria-label="icon: edit"]');
        if (!editBtn1) editBtn1 = document.querySelector('i[aria-label="icon: edit"]');

        if (!editBtn1) {
            const buttons = document.querySelectorAll('button, a[role="button"]');
            for (const btn of buttons) {
                const svg = btn.querySelector('svg[data-icon="edit"], svg[data-icon="pen"]');
                if (svg || btn.title?.toLowerCase().includes('sửa')) {
                    editBtn1 = btn;
                    break;
                }
            }
        }

        if (!editBtn1) {
            throw new Error('Không tìm thấy cây bút chì mở chi tiết câu hỏi (Pencil 1)');
        }

        highlightElement(editBtn1, 'blue');
        simulateFullClick(editBtn1);
        simulateFullClick(editBtn1.querySelector('svg'));
        if (editBtn1.parentElement) simulateFullClick(editBtn1.parentElement);

        // BƯỚC 2: Chờ và click cây bút chì thứ hai (Pencil 2) để mở modal soạn thảo (Edit)
        let editBtn2 = null;
        log("Đang chờ chi tiết câu hỏi hiển thị và tìm Pencil 2...");

        function scoreElement(el, index, totalCount) {
            if (!el) return -999;
            let score = 0;
            const html = el.innerHTML.toLowerCase();
            const className = (typeof el.className === 'string' ? el.className : (el.className?.baseVal || '')).toLowerCase();
            
            // Lấy tất cả thuộc tính có khả năng chứa text tooltip
            const titleAttr = (el.getAttribute('title') || el.title || '').toLowerCase();
            const tooltipAttr = (el.getAttribute('data-original-title') || el.getAttribute('data-tooltip') || el.getAttribute('data-tip') || '').toLowerCase();
            const ariaAttr = (el.getAttribute('aria-label') || '').toLowerCase();
            const parentText = el.parentElement ? (el.parentElement.title || el.parentElement.getAttribute('aria-label') || el.parentElement.getAttribute('data-original-title') || '').toLowerCase() : '';
            
            const allText = `${titleAttr} ${tooltipAttr} ${ariaAttr} ${className} ${parentText}`.toLowerCase();
            
            // 1. Dấu hiệu văn bản edit/sửa cụ thể
            if (allText.includes('chỉnh sửa câu hỏi') || allText.includes('sửa câu hỏi') || allText.includes('edit question')) {
                score += 1000; // Thắng tuyệt đối nếu khớp tooltip cụ thể!
            } else if (allText.includes('edit') || allText.includes('sửa')) {
                score += 100;
            }
            
            // 2. Dấu hiệu hình ảnh bút chì
            if (html.includes('data-icon="edit"') || html.includes('anticon-edit') || html.includes('pencil') || html.includes('pen-to-square') || html.includes('edit-pencil')) {
                score += 80;
            }
            
            // 3. Vị trí: thứ 2 từ phải sang (kế cuối)
            if (totalCount >= 2 && index === totalCount - 2) {
                score += 50;
            }
            
            // Trừ điểm các nút có dấu hiệu Copy
            if (allText.includes('copy') || allText.includes('duplicate') || html.includes('copy') || html.includes('duplicate') || html.includes('copy-o')) {
                score -= 60;
            }
            
            // Trừ điểm các nút có dấu hiệu Xóa/Delete
            if (allText.includes('delete') || allText.includes('xóa') || html.includes('delete') || html.includes('xóa') || html.includes('trash')) {
                score -= 100;
            }
            
            // Trừ điểm các nút đính kèm/attachment (Nút 4, 5 trong danh sách của user)
            if (allText.includes('attach') || allText.includes('file') || html.includes('attach') || html.includes('dropzone')) {
                score -= 80;
            }
            
            return score;
        }

        for (let i = 0; i < 30; i++) { // Chờ tối đa 15 giây (30 * 500ms)
            // Lấy vùng chi tiết đang mở (modal hoặc div.question)
            const detailArea = document.querySelector('.ant-modal-body, .ant-modal, .modal_dialog_ant, div.question, .whitebox');
            if (detailArea) {
                // Ưu tiên tìm trong thanh header metadata chứa các tác vụ ở đầu
                const header = detailArea.querySelector('div.rounded.px-3.py-2, div.rounded:nth-of-type(1)');
                if (header) {
                    const candidates = Array.from(header.querySelectorAll('span.cursor-pointer, .cursor-pointer, i[aria-label="icon: edit"], svg'));
                    const filteredCandidates = candidates.filter(el => el !== editBtn1 && el.offsetWidth > 0);
                    
                    let bestEl = null;
                    let bestScore = -999;
                    filteredCandidates.forEach((el, idx) => {
                        const score = scoreElement(el, idx, filteredCandidates.length);
                        if (score > bestScore) {
                            bestScore = score;
                            bestEl = el;
                        }
                    });
                    
                    if (bestEl && bestScore >= 0) {
                        editBtn2 = bestEl;
                        break;
                    }
                } else {
                    // Fallback nếu không tìm thấy header riêng biệt
                    const candidates = Array.from(detailArea.querySelectorAll('span.cursor-pointer, .cursor-pointer, i[aria-label="icon: edit"], svg'));
                    const filteredCandidates = candidates.filter(el => el !== editBtn1 && el.offsetWidth > 0);
                    
                    let bestEl = null;
                    let bestScore = -999;
                    filteredCandidates.forEach((el, idx) => {
                        const score = scoreElement(el, idx, filteredCandidates.length);
                        if (score > bestScore) {
                            bestScore = score;
                            bestEl = el;
                        }
                    });
                    
                    if (bestEl && bestScore >= 40) {
                        editBtn2 = bestEl;
                        break;
                    }
                }
            }
            if (editBtn2) break;
            await sleep(500);
        }

        if (editBtn2) {
            log("Tìm thấy Pencil 2 (chỉnh sửa tổng), tiến hành click...");
            highlightElement(editBtn2, 'red');
            simulateFullClick(editBtn2);
            
            // Bấm thêm SVG bên trong nếu có
            const svg = editBtn2.querySelector('svg');
            if (svg) simulateFullClick(svg);
            
            if (editBtn2.parentElement) simulateFullClick(editBtn2.parentElement);
            await sleep(1500);
        } else {
            log("Không tìm thấy Pencil 2 tiêu chuẩn, thực hiện click fallback tất cả các bút khác...");
            let pencils = document.querySelectorAll('i[aria-label="icon: edit"]');
            for (let p of pencils) {
                if (p !== editBtn1 && p.offsetWidth > 0) {
                    simulateFullClick(p);
                    if (p.parentElement) simulateFullClick(p.parentElement);
                }
            }
            await sleep(1500);
        }

        // Đợi modal soạn thảo hiển thị hoàn tất (hỗ trợ .wrapper-quill, .ql-editor, .ql-container)
        let formFound = false;
        log("Đang chờ modal soạn thảo load xong...");
        for (let i = 0; i < 30; i++) { // Chờ tối đa 15s (30 * 500ms)
            if (document.querySelector('.wrapper-quill, .ql-editor, .ql-container, div.form-button-groups button.btn-primary')) {
                formFound = true;
                break;
            }
            await sleep(500);
        }

        if (!formFound) {
            throw new Error('Không mở được modal soạn thảo (Không tìm thấy .wrapper-quill, .ql-editor hoặc nút cập nhật sau 15s)');
        }

        // Báo cáo về server
        await postJSON(`${SERVER}/lms/done`, { event: 'lms_edit_done', session: currentSession });
        setBadge('✅ Đã mở Edit', '#10b981');
    }

    // ==========================================
    // POLLING LOOP
    // ==========================================
    async function pollServer() {
        if (isProcessing) return;
        try {
            const res = await postJSON(`${SERVER}/lms/poll?tool=findopen`, {});
            if (res && res.command) {
                await handleCommand(res.command);
            }
        } catch (err) {
            if (badge) setBadge('🔴 Mất kết nối', '#ef4444');
        }
    }

    // ==========================================
    // KHỞI CHẠY
    // ==========================================
    function init() {
        createBadge();
        setInterval(pollServer, POLL_INTERVAL);
        setTimeout(pollServer, 1000);
    }

    if (document.readyState === 'loading') {
        document.addEventListener('DOMContentLoaded', init);
    } else {
        init();
    }
})();
